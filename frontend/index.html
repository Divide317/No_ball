    Python Production SDK Example 
    import requests

url = "http://127.0.0.1:8000/check-delivery"
files = {
    "foot_image": open("crease_frame.jpg", "rb"),
    "waist_image": open("waist_frame.jpg", "rb"),
}
payload = {
    "match_format": "IPL",
    "over": 18,
    "ball_index_in_innings": 112,
    "bouncers_in_over_before": 1,
    "is_current_ball_bouncer": False,
    "fielders_leg_side": 4,
    "fielders_behind_square": 1,
    "fielders_outside_circle": 5,
}
res = requests.post(url, files=files, data=payload).json()
print(res["verdict"], res["ai_overstep_probability"], res["reasons"])
